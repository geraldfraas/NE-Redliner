  for(const p of paras){
    const out=[]; for(const r of p.runs){
      const pc=out.length?out[out.length-1].t.slice(-1):'';                      /* a double space right after a period is real ("Sec. 33.  Sections ..."), keep it; collapse any other run of spaces */
      r.t=r.t.replace(/ {2,}/g,(m,o,s)=>m.length===2&&(o>0?s[o-1]:pc)==='.'?'  ':' ');const l=out[out.length-1];
      if(l&&l.b===r.b&&l.u===r.u&&l.s===r.s&&l.r===r.r)l.t+=r.t;else out.push(r);}
    out.forEach((r,i)=>{if(r.t===' '&&i>0&&i<out.length-1)for(const k of 'usr')if(out[i-1][k]&&out[i+1][k])r[k]=true;});
    const merged=[]; for(const r of out){const l=merged[merged.length-1];
      if(l&&l.b===r.b&&l.u===r.u&&l.s===r.s&&l.r===r.r)l.t+=r.t;else merged.push(r);}
    p.runs=merged;
  }
  if(!paras.length) throw new Error('No numbered text lines were found. This tool reads bills and amendments from the legislature’s drafting system.');
  /* document type */
  const ftxt=front.map(f=>f.t).join(' '), firstText=paras[0].runs.map(r=>r.t).join('');
  let docType='other';
  if(/LEGISLATIVE BILL/i.test(ftxt)||/^A BILL FOR AN ACT/i.test(firstText))docType='bill';
  else if(/AMENDMENT/i.test(ftxt)||/^AM\d+/i.test(running[0]||''))docType='amendment';
  const pre=PRESET[docType], n=pages.length, mid=perPage.slice(1,-1).filter(Boolean);
  const top1=firstTops[0]??pre.firstTop;
  const layout={
    firstTop:top1,
    firstN:docType==='bill'?pageOneCap(top1):(n>1?perPage[0]:pre.firstN),   /* a bill's page 1 is as long as the content makes it (see layout), so what the file happened to hold is not a limit */
    bodyTop:n>1&&firstTops[1]?firstTops[1]:pre.bodyTop,
    perPage:mid.length?Math.max(...mid):pre.perPage,
    enactBreak:docType==='bill'};
  let author='',note='';
  for(const t of footer){const m=t.match(/^Redlined by (.*?)(?: · (.*))?$/); if(m){author=m[1];note=m[2]||'';}}
  return {docType,layout,running,front,paras,author,note};
}

function isRedColor(c){
  if(!c) return false; c=c.trim().toLowerCase(); if(c==='red') return true;
  let m=c.match(/^#([0-9a-f]{6})$/), r,g,b;
  if(m){r=parseInt(m[1].slice(0,2),16);g=parseInt(m[1].slice(2,4),16);b=parseInt(m[1].slice(4,6),16);}
  else{m=c.match(/rgba?\(\s*(\d+)[ ,]+(\d+)[ ,]+(\d+)/); if(!m) return false; r=+m[1];g=+m[2];b=+m[3];}
  return r>140&&g<90&&b<90;
}
function walk(n,st,runs){
  if(n.nodeType===3){ const t=n.nodeValue.replace(/[\r\n\t]+/g,' '); if(t) runs.push({t,b:st.b,u:st.u,s:st.s,r:st.r}); return; }
  if(n.nodeType!==1||n.tagName==='BR') return;
  const s={...st}, tag=n.tagName;
  if(tag==='B'||tag==='STRONG') s.b=true;
  if(tag==='U') s.u=true;
  if(tag==='S'||tag==='STRIKE'||tag==='DEL') s.s=true;
  const fw=n.style&&n.style.fontWeight; if(fw==='bold'||+fw>=600) s.b=true;
  const td=n.style&&(n.style.textDecorationLine||n.style.textDecoration);
  if(td&&td.includes('underline')) s.u=true;
  if(td&&td.includes('line-through')) s.s=true;
  const col=n.getAttribute('color')||(n.style&&n.style.color); if(col) s.r=isRedColor(col);
  if(n.classList&&n.classList.contains('r')) s.r=true;
  n.childNodes.forEach(c=>walk(c,s,runs));
}
function mergeRuns(runs){

}
