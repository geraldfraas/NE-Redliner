/* Words of a model: each has its text (t), the global line it starts on (line) and ends on (end), whether it is struck through (s),
   and whether a space came before it (sp). Trailing , ; : . are split off as their own token, so "approved," -> approved + "," */
function wordsOf(m){
  const raw=[]; let ln=0;
  for(const p of m.paras){
    let cur=null;
    for(const L of wrapPara(p)){
      for(const tk of L.toks){
        if(!cur||tk.gap.length){cur={cells:[],line:ln,end:ln};raw.push(cur);}
        for(const c of tk.cells)cur.cells.push(c);
        cur.end=ln;
      }
      ln++;
    }
  }
  const W=[], mk=(cells,line,end,sp)=>({t:cells.map(c=>c.c).join(''),s:cells.every(c=>c.s),line,end,sp});
  for(const r of raw){
    const t=r.cells.map(c=>c.c).join(''), k=((t.match(/^.*[A-Za-z0-9]([,;:.]+)$/)||[])[1]||'').length;   /* trailing , ; : . after a letter or digit (no lookbehind, which older Safari cannot parse) */
    if(k&&k<t.length){W.push(mk(r.cells.slice(0,t.length-k),r.line,r.end,true));W.push(mk(r.cells.slice(t.length-k),r.end,r.end,false));}
    else W.push(mk(r.cells,r.line,r.end,true));
  }
  return W;
}

/* word alignment (Myers diff on top of a common prefix/suffix trim): map[i] = index in b of a[i], or -1 */
function matchWords(a,b){
  const n=a.length,m=b.length,map=new Int32Array(n).fill(-1);
  let s=0; while(s<n&&s<m&&a[s]===b[s]){map[s]=s;s++;}
  let e=0; while(e<n-s&&e<m-s&&a[n-1-e]===b[m-1-e]){map[n-1-e]=m-1-e;e++;}
  const A=a.slice(s,n-e),B=b.slice(s,m-e),N=A.length,M=B.length;
  if(!N||!M) return map;
  const MAX=Math.min(N+M,3000), off=MAX+1, V=new Int32Array(2*MAX+3), trace=[]; let done=-1;
  for(let d=0;d<=MAX&&done<0;d++){
    trace.push(V.slice(off-d-1,off+d+2));
    for(let k=-d;k<=d;k+=2){
      let x=(k===-d||(k!==d&&V[off+k-1]<V[off+k+1]))?V[off+k+1]:V[off+k-1]+1, y=x-k;
      while(x<N&&y<M&&A[x]===B[y]){x++;y++;}
      V[off+k]=x;
      if(x>=N&&y>=M){done=d;break;}
    }
  }
  if(done<0) return map;                       /* too different to align: the middle counts as replaced */
  let x=N,y=M;
  for(let d=done;d>=0;d--){
    const v=trace[d], g=k=>v[k+d+1], k=x-y, pk=(k===-d||(k!==d&&g(k-1)<g(k+1)))?k+1:k-1, px=g(pk), py=px-pk;
    while(x>px&&y>py){x--;y--;map[s+x]=s+y;}
    x=px;y=py;
  }
  return map;
}
/* the notations, in original page/line order: [{text,page,line}] */
function changeList(m,o){
  let A=owCache.get(o); if(!A){A=wordsOf(o);owCache.set(o,A);}
  const B=wordsOf(m), map=matchWords(A.map(w=>w.t),B.map(w=>w.t)), inB=new Uint8Array(B.length);
  for(const j of map) if(j>=0) inB[j]=1;
  const hunks=[]; let h=null, lastKeep=-1;
  const cur=()=>h||(h={del:[],ins:[],prev:lastKeep,next:-1});
  const flush=nx=>{if(h){h.next=nx;hunks.push(h);} h=null;};
  let i=0,j=0;
  while(i<A.length||j<B.length){
    if(i<A.length&&map[i]<0){cur().del.push(i);i++;}                              /* word gone from the text */
    else if(j<B.length&&!inB[j]){if(!B[j].s)cur().ins.push(j);j++;}               /* new word (new text that is already struck nets out to nothing) */
    else{ if(!A[i].s&&B[j].s)cur().del.push(i); else{flush(i);lastKeep=i;} i++;j++; }  /* same word: newly struck counts as deleted */
  }
  flush(-1);
  const isP=t=>/^[,;:.]+$/.test(t);
  for(const h of hunks){      /* slide a plain deletion right so it starts on a word, not on the comma before it: ", mutual, assessment" -> "mutual, assessment," */
    if(h.ins.length||!h.del.length||!h.del.every(k=>map[k]<0)) continue;
    let g=0; while(g++<60&&isP(A[h.del[0]].t)&&h.next>=0&&h.next<A.length&&A[h.next].t===A[h.del[0]].t&&!hunks.some(x=>x!==h&&x.prev===h.next)){h.del.shift();h.del.push(h.next);h.next++;}
  }
  const at=(a,b)=>{const p=pageLine(o,a),q=pageLine(o,b);
    if(p.page===q.page) return p.line===q.line?`Page ${p.page}, Line ${p.line}`:`Page ${p.page}, Lines ${p.line}-${q.line}`;
    return `Page ${p.page}, Line ${p.line} through Page ${q.page}, Line ${q.line}`;};
  const q=s=>'"'+s+'"', qe=s=>'"'+(/[.,;:]$/.test(s)?s:s+'.')+'"';               /* closing period goes inside the quote, unless the text already ends in punctuation */
  return hunks.map(h=>{
    const dw=h.del.map(k=>A[k]), insT=joinW(h.ins.map(k=>B[k])); let text,pos;
    if(dw.length){
      const both=h.ins.length>0, f=both?q:qe;
      const what=dw.length>14?`beginning with ${q(joinW(dw.slice(0,3)))} through ${f(joinW(dw.slice(-3)))}`:f(joinW(dw));
      pos=at(dw[0].line,dw[dw.length-1].end); text=`${pos}, strike ${what}`+(both?` and insert ${qe(insT)}`:'');
    }else if(h.prev>=0){
      const p=A[h.prev], lead=/^[,;:.]+$/.test(p.t)&&h.prev>0?[A[h.prev-1],p]:[p];     /* anchor on "state," rather than a bare comma */
      pos=at(p.end,p.end); text=`${pos}, after ${q(joinW(lead))} insert ${qe(insT)}`;
    }else{const n=A[h.next]; pos=at(n.line,n.line); text=`${pos}, before ${q(n.t)} insert ${qe(insT)}`;}
    const pl=pageLine(o,dw.length?dw[0].line:(h.prev>=0?A[h.prev].end:A[h.next].line));
    return {text:text.replace(/ /g,' '),page:pl.page,line:pl.line};     /* statute numbers use a non-breaking space before "(1)"; the notation reads with a plain one */
  });
}

function wrapPlain(s,n,ind){
  const out=[]; let line='', first=true;
  const push=t=>{out.push((first?'':ind)+t);first=false;}, lim=()=>first?n:n-ind.length;
  for(let w of s.split(' ')){
    while(w.length>lim()){ if(line){push(line);line='';} push(w.slice(0,lim())); w=w.slice(lim()); }
    if(!line) line=w; else if((line+' '+w).length<=lim()) line+=' '+w; else{push(line);line=w;}
  }
  if(line) push(line);
  return out;
}

      case 'Tc':tc=a[0];break;case 'Tw':tw=a[0];break;case 'Tz':th=a[0]/100;break;case 'TL':tl=a[0];break;case 'Ts':rise=a[0];break;
      case 'Tm':tm=a.slice(0,6);tlm=tm;break;
      case 'Td':tlm=mul([1,0,0,1,a[0],a[1]],tlm);tm=tlm;break;
      case 'TD':tl=-a[1];tlm=mul([1,0,0,1,a[0],a[1]],tlm);tm=tlm;break;
      case 'T*':tlm=mul([1,0,0,1,0,-tl],tlm);tm=tlm;break;
      case 'Tj':show(a[0].str);break;

      case 'RG':stroke=a.slice(0,3);break;case 'G':stroke=[a[0],a[0],a[0]];break;

      case 'l':if(!cur){cur=[[a[0],a[1]]];path=[cur];}else cur.push([a[0],a[1]]);break;
      case 're':rects.push(a.slice(0,4));break;
      case 'W':case 'W*':clip=true;break;
      case 'S':case 's':case 'B':case 'b':
        if(!clip)for(const sp of path)for(let k=0;k+1<sp.length;k++){const p=toDev(...sp[k]),q=toDev(...sp[k+1]);
          if(Math.abs(p[1]-q[1])<0.3)lines.push({x0:Math.min(p[0],q[0]),x1:Math.max(p[0],q[0]),top:H-(p[1]+q[1])/2,color:stroke.slice()});}
        path=[];cur=null;clip=false;rects=[];break;
      case 'f':case 'F':case 'f*':
        if(!clip)for(const r of rects){const p=toDev(r[0],r[1]),q=toDev(r[0]+r[2],r[1]+r[3]);

        path=[];cur=null;clip=false;rects=[];break;
      case 'n':path=[];cur=null;clip=false;rects=[];break;
    }
  }
  return {width:W,height:H,chars,lines};
}

function newDoc(kind){
  const src=clone(TEMPLATES[kind==='bill'?'lb':'am']);

    src.paras=[{style:'hang',runs:[{t:'A BILL FOR AN ACT relating to ____; to amend section ____, Reissue Revised Statutes of Nebraska; and to repeal the original section.',b:false,u:false,s:false,r:false}]},
               {style:'flush',runs:[{t:'Be it enacted by the people of the State of Nebraska,',b:false,u:false,s:false,r:false}]},
               {style:'indent',runs:[{t:'Section 1. ',b:true,u:false,s:false,r:false},{t:'Type or paste the bill text here.',b:false,u:false,s:false,r:false}]}];
  }else{
    src.running=['AM____','LB___','__ - '+today()];
    src.front=src.front.map(f=>f.t.startsWith('AMENDMENTS TO')?{...f,t:'AMENDMENTS TO LB___'}:f);
    src.paras=[{style:'indent',runs:[{t:'1. Strike the original sections and insert the following new sections:',b:false,u:false,s:false,r:false}]}];
  }
  src.docType=kind; src.author=''; src.note=''; src.tracked=false; return src;   /* a blank template has no original to list changes against */
}
function markChanges(){
  const flat=m=>{const W=[];for(const p of m.paras){let cur=[];for(const r of p.runs)for(const c of r.t){if(c===' '){if(cur.length)W.push(cur);cur=[];}else cur.push({c,b:r.b,u:r.u,s:r.s,r:r.r});}if(cur.length)W.push(cur);}return W;};
  const key=w=>w.map(x=>x.c).join('');
  const A=flat(original).map(key);
  const newParas=model.paras.map(p=>({style:p.style,cs:p.runs.flatMap(r=>[...r.t].map(c=>({c,b:r.b,u:r.u,s:r.s,r:r.r})))}));
  const W=[];for(const p of newParas){let cur=[];for(const x of p.cs){if(x.c===' '){if(cur.length)W.push(cur);cur=[];}else cur.push(x);}if(cur.length)W.push(cur);}
  const B=W.map(key), n=A.length, mm=B.length, w=mm+1, dp=new Uint16Array((n+1)*w);
  for(let i=n-1;i>=0;i--)for(let j=mm-1;j>=0;j--)dp[i*w+j]=A[i]===B[j]?dp[(i+1)*w+j+1]+1:Math.max(dp[(i+1)*w+j],dp[i*w+j+1]);
  const keep=new Array(mm).fill(false); let i=0,j=0;
  while(i<n&&j<mm){ if(A[i]===B[j]){keep[j]=true;i++;j++;} else if(dp[(i+1)*w+j]>=dp[i*w+j+1])i++; else j++; }
  W.forEach((wd,k)=>{if(!keep[k])wd.forEach(x=>x.r=true);});
  model.paras=newParas.map(p=>({style:p.style,runs:mergeRuns(p.cs.map(x=>({t:x.c,b:x.b,u:x.u,s:x.s,r:x.r})))}));
  for(const p of model.paras){const rs=p.runs;for(let k=1;k<rs.length-1;k++)if(rs[k].t===' '&&rs[k-1].r&&rs[k+1].r)rs[k].r=true;p.runs=mergeRuns(rs);}
}

function nrRedlineFilename(info, date = new Date()) {
  const bill = String(info || 'bill').replace(/[<>:"/\\|?*\x00-\x1f]/g, '_').replace(/\s+/g, ' ').trim().replace(/[. ]+$/, '') || 'bill';
  const mm = String(date.getMonth() + 1).padStart(2, '0');
  const dd = String(date.getDate()).padStart(2, '0');
  const yy = String(date.getFullYear()).slice(-2);
  return `${bill} Redline ${mm}${dd}${yy}.pdf`;
}
