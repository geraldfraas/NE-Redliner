  });

function wrapPlain(s,n,ind){
  const out=[]; let line='', first=true;
  const push=t=>{out.push((first?'':ind)+t);first=false;}, lim=()=>first?n:n-ind.length;
  for(let w of s.split(' ')){
    while(w.length>lim()){ if(line){push(line);line='';} push(w.slice(0,lim())); w=w.slice(lim()); }
    if(!line) line=w; else if((line+' '+w).length<=lim()) line+=' '+w; else{push(line);line=w;}
  }
  if(line) push(line);
  return out;
}
