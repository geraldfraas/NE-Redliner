=0;
  const flush=last=>{const n=lines.length;lines.push({toks:cur,len,last,cols:colsFor(n),x:xFor(n)});cur=[];len=0;};
  for(const t of toks){
    const add=(cur.length?t.gap.length:0)+t.cells.length;
    if(cur.length&&len+add>colsFor(lines.length)) flush(false);
    len+=(cur.length?t.gap.length:0)+t.cells.length; cur.push(t);
  }

}
function buildSegs(L){
  const toks=L.toks, segs=[]; let x=L.x;
  const nG=toks.slice(1).filter(t=>t.gap.length).length;
  const extra=(!L.last&&nG>0)?((XR-L.x)-L.len*GLYPH)/nG:0;     // fractional justification, like the drafting software
  const push=(text,st,w)=>{segs.push({x,text,w,b:!!st.b,u:!!st.u,s:!!st.s,r:!!st.r});x+=w;};
  toks.forEach((t,i)=>{
    if(i>0&&t.gap.length){
      const g=t.gap, all=k=>g.every(c=>c[k]);
      push(' '.repeat(g.length),{b:all('b'),u:all('u'),s:all('s'),r:all('r')},g.length*GLYPH+extra);
    }
    let run=null;
    for(const c of t.cells){
      if(run&&same(run,c)) run.text+=c.c; else {if(run)push(run.text,run,run.text.length*GLYPH);run={text:c.c,b:c.b,u:c.u,s:c.s,r:c.r};}
    }
    if(run)push(run.text,run,run.text.length*GLYPH);
  });
  return segs;
}
/* A bill's page 1 is content-driven: it holds as many numbered lines as physically fit (the last line starts at 738.1pt, 23 lines
   under the bill cover, 27 under an amendment header), and when the enacting clause lands on page 1 the page ends right after it,
   so Section 1 starts page 2. A long summary that runs onto page 2 has no such break. */
const ENACT=/^\s*Be it enacted by the people of the State of Nebraska/i;
const pageOneCap=top=>Math.floor((738.1-top)/PITCH+0.01)+1;
function layout(m){
  const all=[]; for(const p of m.paras){
    const Ls=wrapPara(p), en=ENACT.test(p.runs.map(r=>r.t).join(''));
    Ls.forEach((L,j)=>all.push({segs:buildSegs(L),x:L.x,enact:en&&j===Ls.length-1}));
  }
  const LY=m.layout, pages=[]; let i=0;
  do{
    const first=!pages.length, n=first?LY.firstN:LY.perPage, top0=first?LY.firstTop:LY.bodyTop, lines=[];
    for(let k=0;k<n&&i<all.length;k++,i++){
      lines.push({num:k+1,top:top0+k*PITCH,x:all[i].x,segs:all[i].segs});
      if(first&&LY.enactBreak&&all[i].enact){i++;break;}
    }
    pages.push({n:pages.length+1,lines});
  }while(i<all.length);
  return pages;
}

let chgOn=true;
/* Words of a model: each has its text (t), the global line it starts on (line) and ends on (end), whether it is struck through (s),
   and whether a space came before it (sp). Trailing , ; : . are split off as their own token, so "approved," -> approved + "," */
function wordsOf(m){
  const raw=[]; let ln=0;
  for(const p of m.paras){
    let cur=null;
    for(const L of wrapPara(p)){
      for(const tk of L.toks){
        if(!cur||tk.gap.length){cur={cells:[],line:ln,end:ln};raw.push(cur);}
        for(const c of tk.cells)cur.cells.push(c);
        cur.end=ln;
      }
      ln++;
    }
  }
  const W=[], mk=(cells,line,end,sp)=>({t:cells.map(c=>c.c).join(''),s:cells.every(c=>c.s),line,end,sp});
  for(const r of raw){
    const t=r.cells.map(c=>c.c).join(''), k=((t.match(/^.*[A-Za-z0-9]([,;:.]+)$/)||[])[1]||'').length;   /* trailing , ; : . after a letter or digit (no lookbehind, which older Safari cannot parse) */
    if(k&&k<t.length){W.push(mk(r.cells.slice(0,t.length-k),r.line,r.end,true));W.push(mk(r.cells.slice(t.length-k),r.end,r.end,false));}
    else W.push(mk(r.cells,r.line,r.end,true));
  }
  return W;
}

/* word alignment (Myers diff on top of a common prefix/suffix trim): map[i] = index in b of a[i], or -1 */
function matchWords(a,b){
  const n=a.length,m=b.length,map=new Int32Array(n).fill(-1);
  let s=0; while(s<n&&s<m&&a[s]===b[s]){map[s]=s;s++;}
  let e=0; while(e<n-s&&e<m-s&&a[n-1-e]===b[m-1-e]){map[n-1-e]=m-1-e;e++;}
  const A=a.slice(s,n-e),B=b.slice(s,m-e),N=A.length,M=B.length;
  if(!N||!M) return map;
  const MAX=Math.min(N+M,3000), off=MAX+1, V=new Int32Array(2*MAX+3), trace=[]; let done=-1;
  for(let d=0;d<=MAX&&done<0;d++){
    trace.push(V.slice(off-d-1,off+d+2));
    for(let k=-d;k<=d;k+=2){
      let x=(k===-d||(k!==d&&V[off+k-1]<V[off+k+1]))?V[off+k+1]:V[off+k-1]+1, y=x-k;
      while(x<N&&y<M&&A[x]===B[y]){x++;y++;}
      V[off+k]=x;
      if(x>=N&&y>=M){done=d;break;}
    }
  }
  if(done<0) return map;                       /* too different to align: the middle counts as replaced */
  let x=N,y=M;
  for(let d=done;d>=0;d--){
    const v=trace[d], g=k=>v[k+d+1], k=x-y, pk=(k===-d||(k!==d&&g(k-1)<g(k+1)))?k+1:k-1, px=g(pk), py=px-pk;
    while(x>px&&y>py){x--;y--;map[s+x]=s+y;}
    x=px;y=py;
  }
  return map;
}
